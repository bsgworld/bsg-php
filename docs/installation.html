Installation Methods

Method Package Use Case
Monorepo (dev) bsg/php-sdk-monorepo Development, testing, get both SDKs
Split - V1 bsg/php-api-sdk-v1 Production, only BSG API
Split - V2 bsg/php-api-sdk Production, only One-API

Method 1: From Monorepo (Both SDKs)

Install both V1 and V2 SDKs from a single package. Best for development and testing.

1Update composer.json

{
    "repositories": [
        {
            "type": "vcs",
            "url": "git@github.com:bsgworld/bsg-php.git"
        }
    ],
    "require": {
        "bsg/php-sdk-monorepo": "dev-monorepo"
    }
}

2Install via Composer

composer update

3Use in your code

<?php
require 'vendor/autoload.php';

// V1 - BSG API
use BSG\Api\V1\Configuration;
use BSG\Api\V1\Api\SMSApi;

$config = Configuration::getDefaultConfiguration()
    ->setHost('https://api.bsg.world')
    ->setApiKey('X-API-KEY', 'your-api-key');

$sms = new SMSApi(null, $config);
$prices = $sms->smsGetPrices();

// V2 - One-API
use BSG\Api\V2\Configuration as ConfigV2;
use BSG\Api\V2\Api\AccountApi;

$configV2 = ConfigV2::getDefaultConfiguration()
    ->setHost('https://one-api.bsg.world')
    ->setApiKey('X-API-KEY', 'your-api-key');

$account = new AccountApi(null, $configV2);
$balance = $account->accountBalance();

Note

Using dev-monorepo requires "minimum-stability": "dev" or specific branch alias. For stable versions, use tags like v1.0.0-beta.

Method 2: Split Packages (Production)

Install only the SDK you need. Recommended for production use.

V1 - BSG API Only

{
    "repositories": [
        {
            "type": "vcs",
            "url": "git@github.com:bsgworld/php-api-sdk-v1.git"
        }
    ],
    "require": {
        "bsg/php-api-sdk-v1": "^1.0"
    }
}
<?php
use BSG\Api\V1\Configuration;
use BSG\Api\V1\Api\SMSApi;
use BSG\Api\V1\Api\ViberApi;
use BSG\Api\V1\Api\HLRApi;
use BSG\Api\V1\Api\MNPApi;

$config = Configuration::getDefaultConfiguration()
    ->setHost('https://api.bsg.world')
    ->setApiKey('X-API-KEY', 'your-api-key');

// Send SMS
$sms = new SMSApi(null, $config);
$result = $sms->smsCreateObjects([
    'destination' => 'phone',
    'originator' => 'BSG',
    'body' => 'Hello from BSG!',
    'msisdn' => '380501234567'
]);

V2 - One-API Only

{
    "repositories": [
        {
            "type": "vcs",
            "url": "git@github.com:bsgworld/php-api-sdk.git"
        }
    ],
    "require": {
        "bsg/php-api-sdk": "^1.0"
    }
}
<?php
use BSG\Api\V2\Configuration;
use BSG\Api\V2\Api\AccountApi;
use BSG\Api\V2\Api\SmsCampaignsApi;

$config = Configuration::getDefaultConfiguration()
    ->setHost('https://one-api.bsg.world')
    ->setApiKey('X-API-KEY', 'your-api-key');

// Get balance
$account = new AccountApi(null, $config);
$balance = $account->accountBalance();

// Send SMS campaign
$campaigns = new SmsCampaignsApi(null, $config);
$result = $campaigns->smsSend([
    'sender' => 'BSG',
    'text' => 'Hello from One-API!',
    'phones' => ['+380501234567']
]);
